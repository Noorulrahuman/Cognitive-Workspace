1. Add Real-Time Streaming Endpoint (SSE) in FastAPI

Currently, POST /chat waits 2–5 seconds for the full Groq LLM response before sending a single JSON payload. Adding POST /chat/stream (or POST /api/v1/chat/stream) with stream=True gives users real-time typewriter output with an immediate ~200ms Time-To-First-Token.

backend/app/main.py or backend/app/api/routes.py (Streaming Implementation) 
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
import json, asyncio

@app.post("/chat/stream")
async def chat_stream(req: ChatRequest):
    """
    Streams LLM tokens using Server-Sent Events (SSE).
    Frontend connects using ReadableStream reader.
    """
    session_id = get_or_create_session(req.session_id)
    history = get_history(session_id)
    history.append({"role": "user", "content": req.message})

    async def token_generator():
        try:
            # Enable streaming in Groq client
            stream = client.chat.completions.create(
                model=settings.GROQ_MODEL,
                messages=history,
                temperature=0.7,
                max_tokens=2048,
                stream=True  # <--- Enable streaming
            )

            full_reply = ""
            for chunk in stream:
                token = chunk.choices[0].delta.content or ""
                if token:
                    full_reply += token
                    # Format as standard SSE data line
                    payload = json.dumps({"token": token})
                    yield f"data: {payload}\n\n"
                    await asyncio.sleep(0)  # yield event loop

            # Save completed turn to conversational history
            history.append({"role": "assistant", "content": full_reply})
            
            # Send [DONE] signal to terminate client reader cleanly
            yield "data: [DONE]\n\n"
        except Exception as err:
            err_payload = json.dumps({"error": str(err)})
            yield f"data: {err_payload}\n\n"

    return StreamingResponse(
        token_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",  # Prevents reverse proxy buffering
        }
    )
